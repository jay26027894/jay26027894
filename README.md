    this.playlist = "🎵 Lo-fi + Synthwave";

    const moods = [
      "🐛 Debugging mysteries",
      "✨ Creating digital magic", 
      "🔥 In the coding zone",
      "🤔 Architecting solutions",
      "🚀 Shipping awesome features",
      "💡 Having eureka moments"
